Get a list of the classes (entity types) used in the named graph kgdocu_movies

SELECT * FROM SPARQL_TABLE('
	SELECT ?class (COUNT(?class) AS ?count)
	FROM <kgdocu_movies>
	WHERE {
		?s <http://www.w3.org/1999/02/22-rdf-syntax-ns#type> ?class .
	}
	GROUP BY ?class ORDER BY DESC(?count)
');

	GROUP BY ?class ORDER BY DESC(?count)

Get triples where object name or object label, if available

SELECT * FROM SPARQL_TABLE('
PREFIX demo: <http://kg.demo.sap.com/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
	SELECT ?movie ?p ?o ?label
	FROM <kgdocu_movies>
	WHERE {
		?movie a demo:Film ;
			demo:title ?movie_title ;
	  		?p ?o.
		OPTIONAL{?o rdfs:label|demo:name ?label .}
		FILTER(?movie_title = "The Matrix")
	}
	ORDER BY DESC(?p)
');

Find movies with the term "matrix" in the title, using a regular (function regex())

SELECT * FROM SPARQL_TABLE('
PREFIX demo: <http://kg.demo.sap.com/>
	SELECT ?movie ?movie_title
	FROM <kgdocu_movies>
	WHERE {
		?movie a demo:Film ;
			demo:title ?movie_title .
		FILTER regex(STR(?movie_title), "matrix","i") .

Get actors acting in the movie "The Matrix"

SELECT * FROM SPARQL_TABLE('
PREFIX demo: <http://kg.demo.sap.com/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
	SELECT ?movie ?actor ?actorLabel
	FROM <kgdocu_movies>
	WHERE {
		?movie a demo:Film ;
	  		demo:title "The Matrix" .
		?actor demo:acted_in ?movie ;
	  		rdfs:label ?actorLabel .
	}
');

SELECT * FROM SPARQL_TABLE('
PREFIX demo: <http://kg.demo.sap.com/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>

FROM <book>
WHERE {
  ?author rdfs:label "Rutherford, Alex" .
  ?book demo:writtenBy ?author ;
        demo:title ?book_title .
}
');


Data Understanding


SELECT * FROM SPARQL_TABLE('
PREFIX demo: <http://kg.demo.sap.com/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
 
SELECT ?book ?book_title ?author ?author_name ?publisher ?publisher_name ?genre_name
FROM <book>
WHERE {
	?author demo:penned ?book;
		rdfs:label ?author_name.
	?book a demo:Book;
		demo:title ?book_title.
	?publisher demo:published ?book;
		rdfs:label ?publisher_name.
	?genre demo:xyz ?book;
		rdfs:label ?genre_name.
	FILTER(STR(?publisher_name) = "Random House").
	FILTER(STR(?genre_name) = "history").
}');


Concatenation 

SELECT * FROM SPARQL_TABLE('
PREFIX demo: <http://kg.demo.sap.com/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
 
SELECT ?author ?author_name (GROUP_CONCAT(DISTINCT STR(?genre_name); separator="; ") as ?genre)
FROM <book>
WHERE {
	?author demo:penned ?book;
		rdfs:label ?author_name.
	?genre demo:xyz ?book;
		rdfs:label ?genre_name.
}
GROUP BY ?author ?author_name
');
