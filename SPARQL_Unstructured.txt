2] Query to display values where subject has string "http://www.example.com/"
SELECT * FROM SPARQL_TABLE('
SELECT ?s ?p ?o
FROM DEFAULT
WHERE {
    ?s ?p ?o .    
    FILTER(
        REGEX(str(?s), "http://www.example.com/")
    )
}
');

3] Get a list of the predicates
SELECT * FROM SPARQL_TABLE('
	SELECT ?p (COUNT(?p) AS ?count)
	FROM DEFAULT
	WHERE {
		?s ?p ?o .
	}
	GROUP BY ?p ORDER BY DESC(?count)
');

4] Get a list of the classes (entity types) used in the named graph 
SELECT * FROM SPARQL_TABLE('
	SELECT ?class (COUNT(?class) AS ?count)
	FROM DEFAULT
	WHERE {
		?s <http://www.w3.org/1999/02/22-rdf-syntax-ns#type> ?class .
	}
	GROUP BY ?class ORDER BY DESC(?count)
');

10] Find the price of services in EUR for service - Microsoft Azure

SELECT ?service ?price
FROM DEFAULT
WHERE {

}
');

15] To return only services that are of type Service or Region
SELECT * FROM SPARQL_TABLE('
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX rdfs: <http://www.example.com/>
SELECT ?service ?type ?name
WHERE {
  ?service rdf:type ?type .
  ?service rdfs:name ?name .
  FILTER(?type IN (<http://www.example.com/Service>, <http://www.example.com/Region>))
}
');


16] Provide services AWS Europe (Frankfurt) with its associated provider Google Cloud or Microsoft Azure
SELECT * FROM SPARQL_TABLE('
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX rdfs: <http://www.example.com/>

SELECT ?service ?serviceName ?provider ?location 
WHERE {
  ?service rdf:type rdfs:Service .
  ?service rdfs:name ?serviceName .
  ?service rdfs:provider ?provider .
  ?service rdfs:locations ?location.
  FILTER(?serviceName IN ("AWS Europe (Frankfurt)"))
  FILTER(?provider IN ("Google Cloud", "Microsoft Azure"))
}
');
