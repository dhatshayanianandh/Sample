SELECT * FROM SPARQL_TABLE('
PREFIX demo: <http://kg.demo.sap.com/>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
	SELECT ?material ?BOM ?Components ?Correlation ?Plant ?AlternateBOM (SUM(xsd:decimal(?quantityValue)) AS ?totalQuantity)
	FROM <AltBOM>
	WHERE {
		?material demo:hasBOM ?BOM.
		?BOM demo:hasComponent ?Components.
		?Components	demo:isappendedto ?Correlation.
		?Correlation demo:isValidFor ?Plant.
		?Correlation demo:hasAlternateBOM ?AlternateBOM.
		?Correlation demo:hasAlternateBOMQuantity ?quantityNode .
				?quantityNode a demo:Quantity ;
                demo:numericValue ?quantityValue .
		FILTER regex(STR(?material), "FG_0024","i") .
	}
	Group by ?material ?BOM ?Components ?Correlation ?Plant ?AlternateBOM
');
